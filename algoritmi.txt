#include <stdio.h>
#include <stdlib.h>
#include <time.h>
#define _CRT_SECURE_NO_WARRNINGS


void gen_arr(float V[], int n, float dg, float gg)
{
    for (int i = 0; i < n; i++)
    {
        V[i] = dg + (float)rand() / RAND_MAX * (gg - dg);
    }
}


int sekv_pret(float V[], int n, float x)
{
    for (int i = 0; i < n; i++)
    {
        if (V[i] == x)
            return i;
    }

    return -1;
}


void merge(float V[], float temp[], int dg, int s, int gg)
{
    int i = dg;
    int j = s + 1; //desni element prve polovice
    int k = dg;

    while (i <= s && j <= gg)
    {
        if (V[i] <= V[j])
            temp[k++] = V[i++];
        else
            temp[k++] = V[j++];
    }

    while (i <= s)
        temp[k++] = V[i++];

    while (j <= gg)
        temp[k++] = V[j++];

    for (i = dg; i <= gg; i++)
        V[i] = temp[i];
}

void merge_sort(float V[], float temp[], int dg, int gg)
{
    if (dg < gg)
    {
        int s = (dg + gg) / 2;

        merge_sort(V, temp, dg, gg);
        merge_sort(V, temp, s + 1, gg);

        merge(V, temp, dg, s, gg);
    }
}

void sort(float V[], int n)
{
    float* temp = malloc(n * sizeof(float));

    if (temp == NULL)
    {
        printf("Greska pri alokaciji memorije!\n");
        exit(1);
    }

    merge_sort(V, temp, 0, n - 1);

    free(temp);
}

int bin_pret(float V[], int n, float x)
{
    int dg = 0;
    int gg = n - 1;

    while (dg <= gg)
    {
        int s =(dg + gg) / 2;

        if (V[s] == x)
            return s;

        if (V[s] < x)
            dg = s + 1;
        else
            gg = s - 1;
    }

    return -1;
}

double vrijeme(clock_t pocetak, clock_t kraj)
{
    return 1000.0 * (kraj - pocetak) / CLOCKS_PER_SEC;
}

int main()
{
    int n;

    printf("Unesite broj elemenata polja: ");
    scanf("%d", &n);

    if (n <= 0)
        return 1;

    float* V = malloc(n * sizeof(float));

    if (V == NULL)
    {
        printf("Nema dovoljno memorije!\n");
        return 1;
    }

    srand((unsigned)time(NULL));

    gen_arr(V, n, 0, 1000);

    float x = 1001;

    clock_t t1, t2;

    t1 = clock();

    int rezultat1 = sekv_pret(V, n, x);

    t2 = clock();

    printf("\nSekvencijalno pretrazivanje: %.3f ms",
        vrijeme(t1, t2));

    t1 = clock();

    sort(V, n);

    t2 = clock();

    printf("\nSortiranje: %.3f ms",
        vrijeme(t1, t2));

    t1 = clock();

    int rezultat2 = bin_pret(V, n, x);

    t2 = clock();

    printf("\nBinarno pretrazivanje: %.3f ms",
        vrijeme(t1, t2));

    printf("\n\nRezultat sekvencijalnog: %d", rezultat1);
    printf("\nRezultat binarnog: %d\n", rezultat2);

    free(V);

    return 0;
}
